     5→// Variant prices are exactly proportional to the per-kg price from the
     6→// client's price list. MRP carries a ~20% markup rounded to the nearest 10.
     7→function pickleVariants(pricePerKg: number) {
     8→  const r10 = (n: number) => Math.round(n / 10) * 10;
     9→  return [
    10→    { id: "v-250", label: "250 g", grams: 250, price: r10(pricePerKg / 4), mrp: r10((pricePerKg / 4) * 1.2) },
    11→    { id: "v-500", label: "500 g", grams: 500, price: r10(pricePerKg / 2), mrp: r10((pricePerKg / 2) * 1.2) },
    12→    { id: "v-1000", label: "1 kg", grams: 1000, price: pricePerKg, mrp: r10(pricePerKg * 1.2) },
    13→  ];
    14→}
